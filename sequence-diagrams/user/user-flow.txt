== User Registration Flow ==
Client -> Router: POST /registration\n{name, email, password}
Router -> Controller: registrationUser()
Controller -> Model: findOne({email})

alt Email already exists
    Model --> Controller: user exists
    Controller -> Error: "Email already exist", 400
    Error --> Client: 400 Error Response
else Email available
    Model --> Controller: null
    Controller -> Controller: createActivationToken(user)
    note over Controller: Generate 4-digit code &\nJWT token (5m expiry)
    Controller -> Controller: Math.floor(1000 + Math.random() * 9000)
    Controller -> JWT: sign({user, activationCode},\nACTIVATION_SECRET, 5m)
    JWT --> Controller: activation token
    Controller -> Email: renderFile("activation-mail.ejs", data)
    Email --> Controller: HTML template rendered
    Controller -> Email: sendMail({email, subject, template, data})
    
    alt Email send success
        Email --> Controller: email sent successfully
        Controller --> Client: 201 {success: true, message, activationToken}
    else Email send failure
        Email --> Controller: email error
        Controller -> Error: error.message, 400
        Error --> Client: 400 Error Response
    end
end

== User Activation Flow ==
Client -> Router: POST /activate-user\n{activation_token, activation_code}
Router -> Controller: activateUser()
Controller -> JWT: verify(activation_token, ACTIVATION_SECRET)

alt Invalid or expired token
    JWT --> Controller: verification failed
    Controller -> Error: error.message, 400
    Error --> Client: 400 Error Response
else Valid token
    JWT --> Controller: {user: IUser, activationCode: string}
    Controller -> Controller: compare codes
    
    alt Invalid activation code
        Controller -> Error: "Invalid activation Code", 400
        Error --> Client: 400 Error Response
    else Valid activation code
        Controller -> Model: findOne({email}) - check if user exists
        
        alt User already exists
            Model --> Controller: existing user
            Controller -> Error: "Email already Exist", 400
            Error --> Client: 400 Error Response
        else User doesn't exist
            Model --> Controller: null
            Controller -> Model: create({name, email, password})
            note over Model: Password hashed in pre-save hook
            Model -> Model: bcrypt.hash(password, 10)
            Model --> Controller: user created & saved
            Controller --> Client: 201 {success: true}
        end
    end
end

== User Login Flow ==
Client -> Router: POST /login {email, password}
Router -> Controller: loginUser()

alt Missing credentials
    Controller -> Error: "Please enter email and password", 400
    Error --> Client: 400 Error Response
else Valid input
    Controller -> Model: findOne({email}).select("+password")
    
    alt User not found
        Model --> Controller: null
        Controller -> Error: "Invalid email or password", 400
        Error --> Client: 400 Error Response
    else User found
        Model --> Controller: user with password
        Controller -> Model: comparePassword(password)
        Model -> Model: bcrypt.compare(password, this.password)
        
        alt Password mismatch
            Model --> Controller: false
            Controller -> Error: "Invalid email or password", 400
            Error --> Client: 400 Error Response
        else Password match
            Model --> Controller: true
            Controller -> Controller: sendToken(user, 200, res)
            Controller -> Model: SignAccessToken() & SignRefreshToken()
            Model -> JWT: sign({id: this._id}, ACCESS_TOKEN, 5m)
            JWT --> Model: access token
            Model -> JWT: sign({id: this._id}, REFRESH_TOKEN, 3d)
            JWT --> Model: refresh token
            Model --> Controller: {accessToken, refreshToken}
            Controller -> Redis: set(user._id, JSON.stringify(user), "EX", 604800)
            Redis --> Controller: user cached (7 days)
            Controller --> Client: 200 {success, user} + Set Cookies
            note over Client: Cookies: access_token, refresh_token
        end
    end
end

== Authentication Middleware Flow ==
Client -> Router: Any Protected Route (with cookies)
Router -> Auth: isAuthenticated()
Auth -> Auth: extract access_token from cookies

alt No access token
    Auth -> Error: "Please login to access this resource", 400
    Error --> Client: 400 Error Response
else Access token exists
    Auth -> JWT: decode(access_token, {complete: true})
    
    alt Invalid token format
        JWT --> Auth: null
        Auth -> Error: "access token is not valid", 400
        Error --> Client: 400 Error Response
    else Valid token format
        JWT --> Auth: decoded token with exp
        Auth -> Auth: check if token expired
        
        alt Token expired (decoded.exp <= Date.now()/1000)
            Auth -> Controller: updateAccessToken(req, res, next)
            Controller -> Controller: extract refresh_token from cookies
            Controller -> JWT: verify(refresh_token, REFRESH_TOKEN)
            
            alt Invalid refresh token
                JWT --> Controller: verification failed
                Controller -> Error: "Could not refresh token", 400
                Error --> Client: 400 Error Response
            else Valid refresh token
                JWT --> Controller: decoded refresh token
                Controller -> Redis: get(decoded.id)
                
                alt No session in Redis
                    Redis --> Controller: null
                    Controller -> Error: "Please login for access this resources!", 400
                    Error --> Client: 400 Error Response
                else Session exists
                    Redis --> Controller: user session JSON
                    Controller -> Controller: JSON.parse(session)
                    Controller -> JWT: sign({id: user._id}, ACCESS_TOKEN, 5m)
                    JWT --> Controller: new access token
                    Controller -> JWT: sign({id: user._id}, REFRESH_TOKEN, 3d)
                    JWT --> Controller: new refresh token
                    Controller -> Controller: req.user = user
                    Controller -> Controller: set cookies with new tokens
                    Controller -> Redis: set(user._id, JSON.stringify(user), "EX", 604800)
                    Redis --> Controller: session refreshed
                    Controller --> Auth: next() - continue
                end
            end
        else Token still valid
            Auth -> Redis: get(decoded.payload.id)
            
            alt No user in Redis
                Redis --> Auth: null
                Auth -> Error: "Please login to access this resource", 400
                Error --> Client: 400 Error Response
            else User found in Redis
                Redis --> Auth: user JSON
                Auth -> Auth: req.user = JSON.parse(user)
                Auth --> Router: next() - continue
            end
        end
    end
end

== Authorization Middleware Flow ==
Router -> Auth: authorizeRoles(...roles)
Auth -> Auth: extract user from req.user
Auth -> Auth: const role = user?.role
Auth -> Auth: check if roles.includes(role)

alt Role not authorized
    Auth -> Error: `Role: ${role} is not allowed\nto access this resource`, 403
    Error --> Client: 403 Forbidden
else Role authorized
    Auth --> Router: next() - continue
end

== Update Password Flow ==
Client -> Router: PUT /update-user-password\n{oldPassword, newPassword}
Router -> Auth: isAuthenticated()
Auth --> Router: user authenticated
Router -> Controller: updatePassword()

alt Missing passwords
    Controller -> Error: "Please enter old and new password", 400
    Error --> Client: 400 Error Response
else Passwords provided
    Controller -> Controller: extract user from req.user
    Controller -> Model: findById(id).select("+password")
    Model --> Controller: user with password
    
    alt User password undefined
        Controller -> Error: "Invalid user", 400
        Error --> Client: 400 Error Response
    else Valid user
        Controller -> Model: comparePassword(oldPassword)
        Model --> Controller: password comparison result
        
        alt Password mismatch
            Controller -> Error: "Invalid old password", 400
            Error --> Client: 400 Error Response
        else Password match
            Controller -> Controller: user.password = newPassword
            Controller -> Model: save()
            Model -> Model: bcrypt.hash(newPassword, 10) in pre-save
            Model --> Controller: password updated
            Controller -> Redis: set(id, JSON.stringify(user))
            Redis --> Controller: cache updated
            Controller --> Client: 201 {success: true, user}
        end
    end
end
